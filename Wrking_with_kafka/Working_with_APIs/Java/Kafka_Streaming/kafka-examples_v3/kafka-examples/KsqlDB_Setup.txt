CREATE STREAM input_text (
    LINE VARCHAR
) WITH (
    KAFKA_TOPIC='streams-plaintext-input',
    VALUE_FORMAT='KAFKA'
);

--Declare a ksqlDB TABLE over the output topic
our app writes key = word (String), value = count (Long, via Serdes.Long()). 
ksqlDB's KAFKA format supports raw primitive types directly — no Avro/JSON needed — so we can map it 1:1
CREATE TABLE word_counts (
    WORD VARCHAR PRIMARY KEY,
    WORD_COUNT BIGINT
) WITH (
    KAFKA_TOPIC='streams-wordcount-output',
    VALUE_FORMAT='KAFKA',
    KEY_FORMAT='KAFKA'
);

Note**A TABLE (not STREAM) here since the topic is a changelog — 
each new record for a word is the latest total, not an independent event.

--push query
SELECT * FROM word_counts EMIT CHANGES;

(if it shows errors:
KAFKA uses serialization format — it only knows how to encode a single value, 
not a multi-field record. our table only has one real value column (WORD_COUNT), But SELECT * on a table implicitly 
adds ksqlDB's pseudocolumns ROWPARTITION and ROWOFFSET to the result. 
Once those are added, the result becomes a multi-field schema, and KAFKA format can't serialize that.)

--so this wouldnt work too.
ksql> SELECT WORD, WORD_COUNT FROM word_counts EMIT CHANGES;

when we push-query a source table (one declared directly over an existing topic via plain CREATE TABLE ... 
WITH (KAFKA_TOPIC=...)), ksqlDB internally tracks ROWPARTITION/ROWOFFSET as part of how it processes 
the underlying changelog for that table — regardless of what we put in your SELECT clause. 
That metadata gets folded into the schema that has to be serialized, and KAFKA format can only serialize a single bare field. 
So any push query directly against this raw word_counts table will hit the same wall, no matter how we write the SELECT.
CTAS inherits the source table's value format (KAFKA) by default unless we explicitly override.

--Query current state (a snapshot like a normal database)
SELECT * FROM word_counts WHERE WORD = 'hello';

(if error:
A table declared with a plain CREATE TABLE ... WITH (KAFKA_TOPIC=...) (a source table) is  a logical schema over an existing topic
 — ksqlDB doesn't build its own materialized state store for it. 
Pull queries (SELECT ... WHERE ... without EMIT CHANGES) need that materialized store to look up by key, 
so they only work against tables that ksqlDB itself materializes — i.e., ones built via CREATE TABLE ... AS SELECT (CTAS).)

CREATE TABLE queryable_word_counts AS

#Changes in Appl
--switch app's sink to JSON
--Change the output value serde from raw Serdes.Long() to a small JSON-encoded object (so ksqlDB can map a named column onto it):

import com.fasterxml.jackson.databind.ObjectMapper;
import org.apache.kafka.common.serialization.Serde;
import org.apache.kafka.common.serialization.Serdes;
import org.apache.kafka.common.serialization.Serializer;
import org.apache.kafka.common.serialization.Deserializer;
import java.util.Collections;
import java.util.Map;

static Serde<Long> jsonLongValueSerde() {
    ObjectMapper mapper = new ObjectMapper();

    Serializer<Long> serializer = (topic, data) -> {
        try {
            Map<String, Long> wrapped = Collections.singletonMap("WORD_COUNT", data);
            return mapper.writeValueAsBytes(wrapped);
        } catch (Exception e) { throw new RuntimeException(e); }
    };

    Deserializer<Long> deserializer = (topic, bytes) -> {
        try {
            Map<String, Object> m = mapper.readValue(bytes, Map.class);
            return ((Number) m.get("WORD_COUNT")).longValue();
        } catch (Exception e) { throw new RuntimeException(e); }
    };

    return Serdes.serdeFrom(serializer, deserializer);
}

--update the sink
counts.toStream()
    .to(OUTPUT_TOPIC, Produced.with(Serdes.String(), jsonLongValueSerde()));
